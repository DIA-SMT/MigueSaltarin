"y" es la superficie donde se pisa; el mesh queda debajo. function activarNube(x, y, ancho, vx, esRoca) { const n = nubes.find(n => !n.activa); if (!n) return null; n.activa = true; n.x = x; n.y = y; n.mitad = ancho / 2; n.vx = vx; n.rebote = 0; n.esRoca = !!esRoca; if (n.esRoca) { n.mesh.geometry = geoRocas[(nubes.indexOf(n) + Math.floor(Math.abs(y))) % geoRocas.length]; n.mesh.material = matRoca; n.escalaY = 0.55; n.mesh.scale.set(ancho / ANCHO_GEO_ROCA, n.escalaY, 0.8); n.mesh.position.set(x, y - 0.62, 0); } else { n.mesh.geometry = geoNube; n.mesh.material = matNube; n.escalaY = 0.75; n.mesh.scale.set(ancho / ANCHO_GEO_NUBE, n.escalaY, 0.8); n.mesh.position.set(x, y - 0.45, 0); } n.mesh.visible = true; return n; } // Obstáculo: tormenta (o agujero negro si esAgujero). Misma colisión en ambos casos. function activarTormenta(x, y, esAgujero) { const t = tormentas.find(t => !t.activa); if (!t) return null; t.activa = true; t.x = x; t.y = y; t.esAgujero = !!esAgujero; t.nube.visible = t.rayo.visible = !t.esAgujero; t.agujero.visible = t.esAgujero; t.grupo.visible = true; t.grupo.position.set(x, y, 0); return t; } function desactivarTodo() { for (const n of nubes) { n.activa = false; n.mesh.visible = false; } for (const t of tormentas) { t.activa = false; t.grupo.visible = false; } } // Recicla lo que quedó por debajo de la pantalla (así se reutiliza arriba) function reciclar() { const limite = J.camY - altoVisible / 2 - 3; for (const n of nubes) if (n.activa && n.y < limite) { n.activa = false; n.mesh.visible = false; } for (const t of tormentas) if (t.activa && t.y < limite) { t.activa = false; t.grupo.visible = false; } }
